```python

ruta_origen_hdfs = "/datalake/renfe/raw/estaciones/estaciones.csv"
ruta_destino_hdfs = "/datalake/renfe/processed/estaciones_filtradas.csv"
provincia_elegida = "León"  # Cambia por la provincia que necesites

print(f"Procesando y guardando estaciones para la provincia: {provincia_elegida}...")

# 1. Abrimos el stream de lectura desde HDFS
with client.read(hdfs_path=ruta_origen_hdfs, encoding="latin-1") as reader:
    
    # 2. Abrimos el stream de escritura en HDFS con overwrite=True
    # Nota: Dependiendo de la librería de HDFS, 'client.write' acepta 'encoding' o requiere escribir strings/bytes.
    with client.write(hdfs_path=ruta_destino_hdfs, overwrite=True, encoding="latin-1") as writer:
        
        # Leemos la cabecera
        cabecera = reader.readline()
        if isinstance(cabecera, bytes):
            cabecera = cabecera.decode("latin-1", errors="ignore")
            
        # Escribimos la cabecera en el archivo de destino
        writer.write(cabecera)
        
        contador_filtradas = 0

        # 3. Iteramos línea a línea en streaming y filtramos
        for linea in reader:
            if isinstance(linea, bytes):
                linea_str = linea.decode("latin-1", errors="ignore")
            else:
                linea_str = linea

            # Filtrar por la provincia elegida
            if provincia_elegida.lower() in linea_str.lower():
                writer.write(linea_str)
                contador_filtradas += 1

print(f"¡Proceso completado! Se han guardado {contador_filtradas} líneas filtradas en: {ruta_destino_hdfs}")
            

```

    with client.read(ruta_hdfs, encoding='latin-1', delimiter='\n') as reader:

            print(f"Cabecera: {cabecera.strip()}")

def reset_datalake():
    """
    Elimina por completo y de forma recursiva la estructura /datalake/renfe 
    en HDFS para permitir reiniciar el entorno de trabajo desde cero.
    """
    # 1. Configurar el cliente InsecureClient
    client = InsecureClient('http://namenode:9870', user='root')
    
    # Ruta base del datalake que se desea limpiar
    ruta_datalake = '/datalake/renfe'
    
    try:
        print(f"Eliminando la estructura recursiva en: {ruta_datalake}...")
        
        # 2. Utilizar el método delete con recursive=True
        client.delete(ruta_datalake, recursive=True)
        
        print(f"¡Entorno reiniciado con éxito! Se ha eliminado {ruta_datalake}.")
        
    except Exception as e:
        print(f"Error al intentar eliminar el directorio {ruta_datalake}: {e}")
